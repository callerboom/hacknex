## TEST 3: TTS Audio Generation — RUNS WITHOUT ANY MODEL (sine wave stub)
# =====================================================
# What it tests: synth_stream(), audio chunks, TTFA latency

python -c "
import sys, time
sys.path.insert(0, '.')
import numpy as np
from src.edge.tts import SentenceChunkedTTS

tts = SentenceChunkedTTS()
print(f'Backend: {tts._backend_type} (stub=no model, sherpa_onnx=real TTS)')

sentences = [
    'Cool water should be run over a minor burn for ten to twenty minutes.',
    'Do not apply ice. Consult a doctor for serious burns.',
]

t0 = time.perf_counter()
ttfa_ms = None
chunks = 0
for sent in sentences:
    for chunk in tts.synth_stream(sent):
        if ttfa_ms is None:
            ttfa_ms = (time.perf_counter() - t0) * 1000
        chunks += 1
        assert chunk.dtype == np.float32

total_ms = (time.perf_counter() - t0) * 1000
print(f'TTFA  : {ttfa_ms:.1f} ms   (target < 200ms)')
print(f'Total : {total_ms:.0f} ms')
print(f'Chunks: {chunks}')
print('TTS TEST DONE!')
"

## EXPECTED OUTPUT:
## Backend: stub (or sherpa_onnx if model installed)
## TTFA  : ~3-75 ms   <- PASS if < 200ms

## TEST 4: LLM Streaming — needs llama-cpp-python (stub if not installed)
# =====================================================
# What it tests: token streaming, TTFT latency, response quality

python -c "
import sys, time
sys.path.insert(0, '.')
from src.edge.llm import GGUFLocalLLM

llm = GGUFLocalLLM()
llm.load()

query = 'How do I treat a minor burn?'
print(f'Query: {query}')
print('Response: ', end='', flush=True)

t0 = time.perf_counter()
ttft_ms = None
resp = ''
for tok in llm.stream(query, []):
    if ttft_ms is None:
        ttft_ms = (time.perf_counter() - t0) * 1000
    resp += tok
    print(tok, end='', flush=True)

total_ms = (time.perf_counter() - t0) * 1000
print()
print(f'TTFT : {ttft_ms:.1f} ms')
print(f'Total: {total_ms:.0f} ms')
print(f'Chars: {len(resp)}')
print('LLM TEST DONE!')
"
